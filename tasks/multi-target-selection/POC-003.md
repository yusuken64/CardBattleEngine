```csharp
[TestMethod]
public void AttackTriggerSelectMinionDealsThreeDamage()
{
    var state = GameFactory.CreateTestGame(); // match whatever the existing test fixtures actually use — check CombatTest.cs for the real helper name and how attacks are already driven in this test suite
    var engine = new GameEngine();
    var attackerOwner = state.CurrentPlayer;
    var defenderOwner = state.Players.First(p => p != attackerOwner);

    var attackerCard = new MinionCard("Sniper", 3, 2, 3);
    attackerCard.TriggeredEffects.Add(new TriggeredEffect
    {
        EffectTrigger = EffectTrigger.Attack,
        EffectTiming = EffectTiming.Post,
        Scope = TriggerScope.Self,
        TargetRequirement = new TargetRequirement
        {
            Provider = new EntityTypeSelector { EntityTypes = EntityType.Minion, TeamRelationship = TeamRelationship.Any },
            Count = 1,
        },
        GameActions = [new DamageAction { Damage = (Value)3 }],
    });

    var attacker = new Minion(attackerCard, attackerOwner);
    attacker.HasCharge = true; // or however this test suite's existing CombatTest.cs fixtures make a freshly-summoned minion able to attack immediately
    attackerOwner.Board.Add(attacker);

    var bystander = new Minion(new MinionCard("Bystander", 1, 1, 5), defenderOwner);
    var victim = new Minion(new MinionCard("Victim", 1, 1, 5), defenderOwner);
    defenderOwner.Board.Add(bystander);
    defenderOwner.Board.Add(victim);

    // Drive the attack exactly as CombatTest.cs already does for ordinary attacks in this codebase.
    var attackChoice = state.GetValidActions(attackerOwner).First(a => a.Item1 is AttackAction aa && aa.Attacker == attacker && aa.Defender == bystander);
    engine.Resolve(state, attackChoice.Item2, attackChoice.Item1);

    // The Post-Attack trigger's DamageAction now has a PendingTargetRequirement and no Targets yet,
    // so GameEngine.Resolve's central interception (NTR-005) should have paused here.
    Assert.IsInstanceOfType(state.PendingChoice, typeof(TargetSelectionChoice));

    var pick = state.GetValidActions(attackerOwner).First(a => ((SupplyTargetAction)a.Item1).Candidate == victim);
    engine.Resolve(state, pick.Item2, pick.Item1);

    Assert.AreEqual(2, victim.Health);   // 5 - 3 damage
    Assert.AreEqual(5, bystander.Health); // untouched — the player chose victim, not bystander
}
```
Adapt helper names, constructors, and how this test suite's existing `CombatTest.cs` already drives attacks (charge/summoning-sickness handling, `AttackAction` construction, `GameEngine.Resolve` call shape) to match the real, already-migrated conventions in this codebase. The important behavioral assertions to preserve are: (a) the attack itself resolves normally (attacker/defender combat damage happens as usual — this test only adds a case on top of it), (b) after the attack, the engine pauses with a `TargetSelectionChoice` rather than resolving the triggered `DamageAction` immediately, (c) both minions on the defending board are legal candidates (the trigger's `TargetRequirement.Provider` doesn't restrict to "the defender" — it's a genuinely free player choice), and (d) only the picked minion takes the 3 damage.
